JavaScriptのfor文

for文

for文は、普通のループ文で、VBAのような指定の仕方になっています

以下の例は、numbers[]配列の要素をインデクス0から順番に表示しています。numbers[index] で要素を指定しておりイメージ通りかと思います


          //for文
          const numbers = [10, 20, 30, 40];
          // 初期値; forが回る条件; 増分 で指定する
          for (let index = 0; index < 4; index++) {
            console.log(numbers[index]);
          }
          //実行結果（> node .\hoge.js）
          10
          20
          30
          40

for in文

これは最初知った時に「え？」となったfor文。最初は他の言語のForEach文と同じようなものと思っていました、また他の言語のForEach文と同じような動作をします。つまり、インデクスを指定せずに配列などのコレクションから１つずつ取り出すという動作ですが、違うのは「取り出すのは要素ではなくindex番号」です

以下の例を見ると、動作はForEachだけど取り出しているのが要素ではなく、index値となっていることがわかります


          // person[]配列には、name,age,cityの各プロパティを持つオブジェクトが要素
          const person = [
            { name: "HouNeko", age: 100, city: "Kobe" },
            { name: "HouUsagi", age: 120, city: "Osaka" }
          ];

          // for in 文で、person[]配列から1つずつデータをkeyに取り出している
          // このkey変数に読み込まれるのは「index値」
          for (let key in person) {
            console.log(
              "key = " + key + "\n",
              "person[key].name = " + person[key].name + "\n",
              "person[key].age = " + person[key].age + "\n",
              "person[key].city = " + person[key].city + "\n"
            );
          }          
          //実行結果（> node .\hoge.js）
          kye = 0   //← keyに読み込まれた値は、indexになっていることがわかる
          person[key].name = HouNeko    //keyはindexなので、person[]配列をkey値で参照できる
          person[key].age = 100
          person[key].city = Kobe

          kye = 1
          person[key].name = HouUsagi
          person[key].age = 120
          person[key].city = Osaka

for of文

for of文が、ForEach文に近いもの。for in文はindex値が取り出されるけど、for of文は要素が取り出されます


          // person[]配列には、name,age,cityの各プロパティを持つオブジェクトが要素
          const person = [
            { name: "HouNeko", age: 100, city: "Kobe" },
            { name: "HouUsagi", age: 120, city: "Osaka" }
          ];

          // for of文は、ForEach文みたいなもの
          // val変数には、person[]配列の要素が1つずつ読み出される
          for (let val of person) {
            console.log(
              "val = ", val,
              "\nval.name = ", val.name,
              "\nval.age = ", val.age,
              "\nval.city = ", val.city
            );
          }

          //この実行結果
          > node .\hoge.js
          val =  { name: 'HouNeko', age: 100, city: 'Kobe' }    ← val変数にはperson[]配列の要素が読み出されている
          val.name =  HouNeko
          val.age =  100
          val.city =  Kobe
          val =  { name: 'HouUsagi', age: 120, city: 'Osaka' }
          val.name =  HouUsagi
          val.age =  120
          val.city =  Osaka
