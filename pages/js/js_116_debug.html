開発者ツールを使ったデバッグ

ブラウザ上で動かすJavaScriptのデバッグはブラウザについている開発者ツールで行う事ができる。

console.logの有効な使い方

console.log

JavaScriptをコーディング中は、他の言語と同様、都度修正したり設定を確認したりすると思います。各設定値などをログ出力して確認するけど、本番リリースではログを取らないとだめと思います。この場合、「console.log」をコメントアウトするのは非効率

ログをオン・オフするスイッチを作り、これで出力を制御するようにすればいい


        // ---------------------------------------------------------------------------
        // ログ処理
        //  フラグ： ログ出力を制御用、trueでログ出力。本番リリース時には falseにする
        //  関数  ： args引数の型と値を表示する
        const MYDEBUG = true;                 // このフラグでオン・オフを切り替える
        function debugLog(...args) {
          if (!MYDEBUG) return;               // オフの場合も関数にはロ入るがリターンするだけ
                                              // 数千程度のコールであれば微小なので、速度低下は気にしなくていい
          args.forEach((arg, index) => {
            let type = typeof arg;
            let display = arg;

            if (Array.isArray(arg)) type = 'array';
            if (arg === null) type = 'null';

            console.log(`[arg${index}] type: ${type}, value:`, display);
          });
        }        

        // ---------------------------------------------------------------------------
        // ログ処理をコールする各関数
        function hoge() {
          ...
          debugLog("ログ１です");     // デバッグ中も本番リリース時も子のコードはそのままでいい
          ...                         // MYDEBUG定数を、trueにするかfalseにするがでログ出力を出したり
          ...                         // 出さなかったりできる
          debugLog("ログ２です");
          ...
          ...
          debugLog("ログ３です");
          ...
        }

ログ出力のサンプル例

この例は、detailsタグをネストさせた目次（の中にも目次がある）に対して、一度に全部のdetailsタグを開く・閉じるのボタンを作った時のものです。一度に開・閉する時のJavaScriptの関数の引数で、渡されたdetailsタグが全部あるかを確認した画面です

以下はログ処理部分です


        // ---------------------------------------------------------------------------
        // ログ処理
        //  フラグ： ログ出力を制御用、trueでログ出力。本番リリース時には falseにする
        //  関数  ： args引数の型と値を表示する
        const MYDEBUG = true; 
        function debugLog(...args) {
          if (!MYDEBUG) return;

          args.forEach((arg, index) => {
            let type = typeof arg;
            let display = arg;

            if (Array.isArray(arg)) type = 'array';
            if (arg === null) type = 'null';

            console.log(`[arg${index}] type: ${type}, value:`, display);
          });
        }

以下は関数側の処理です


        function initDetailsOpenClose() {
          const detailsList = document.querySelectorAll(".tocall");
          const openBtn = document.getElementById("details-openAll");
          const closeBtn = document.getElementById("details-closeAll");
          debugLog('details-openclose loaded', detailsList);    // ← ログコール、文字と配列データを渡している

          if (!detailsList.length || !openBtn || !closeBtn) return;

          openBtn.addEventListener("click", () => {
            detailsList.forEach(details => {
              details.open = true;
              debugLog('All details opened');     // ログコール、文字だけ渡している
            });
          });

          closeBtn.addEventListener("click", () => {
            detailsList.forEach(details => {
              details.open = false;
              debugLog('All details closed');     // ログコール、文字だけ渡している
            });
          });
        }
